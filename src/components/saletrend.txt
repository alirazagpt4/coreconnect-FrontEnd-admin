import React, { useMemo } from 'react';
import Chart from 'react-apexcharts';
import { Box, Typography } from '@mui/material';

const SalesTrendChart = ({ data, categories }) => {
    const brandColors = {
        'RIVAJ': '#f59e0b',
        'AMRIJ': '#a855f7',
        'RHD': '#ec4899',
        'NO!MO!': '#10b981',
        'EVERNOYA': '#3b82f6',
    };

    const priorityOrder = ['RIVAJ', 'AMRIJ', 'RHD', 'NO!MO!', 'EVERNOYA'];

    // 1. Sorting logic (Same as before)
    const sortedCategories = useMemo(() => {
        return [...categories].sort((a, b) => priorityOrder.indexOf(a) - priorityOrder.indexOf(b));
    }, [categories]);

    // 2. Prepare Data for ApexCharts
    // ApexCharts mangta hai: series: [{ name: 'Brand', data: [val1, val2] }]
    const series = useMemo(() => {
        return sortedCategories.map(brand => ({
            name: brand,
            data: data.map(item => item[brand] || 0)
        }));
    }, [data, sortedCategories]);

    const dates = useMemo(() => data.map(item => item.date), [data]);

    // 3. Chart Configuration
    const options = {
        chart: {
            type: 'bar',
            stacked: false, // Groups bars side by side
            toolbar: { show: false },
            zoom: { enabled: false },
            animations: { enabled: true, easing: 'easeinout', speed: 800 },
            // YE HAI ASLI ILAJ:
            redrawOnParentResize: true,
            redrawOnWindowResize: true
        },
        colors: sortedCategories.map(brand => brandColors[brand]),
        plotOptions: {
            bar: {
                horizontal: false,
                columnWidth: '55%', // Bars ki thickness yahan se control hoti hai
                borderRadius: 4,
                endingShape: 'rounded'
            },
        },
        dataLabels: { enabled: false },
        stroke: { show: true, width: 2, colors: ['transparent'] },
        xaxis: {
            categories: dates,
            axisBorder: { show: false },
            axisTicks: { show: false },
            labels: { style: { colors: '#94a3b8', fontSize: '12px' } }
        },
        yaxis: {
            labels: {
                style: { colors: '#94a3b8', fontSize: '12px' },
                formatter: (val) => new Intl.NumberFormat('en-US').format(val)
            }
        },
        legend: {
            position: 'bottom',
            fontWeight: 700,
            fontSize: '12px',
            labels: { colors: '#64748b' },
            markers: { radius: 12 }
        },
        grid: { borderColor: '#f1f5f9', strokeDashArray: 4 },
        tooltip: { y: { formatter: (val) => `${val.toLocaleString()}` } }
    };

    if (!data || data.length === 0) return (
        <Box sx={{ display: 'flex', justifyContent: 'center', alignItems: 'center', height: '100%' }}>
            <Typography variant="body2" color="textSecondary">No Data Available</Typography>
        </Box>
    );

    return (
        <Box sx={{ width: '100%', height: '100%', minHeight: '400px' }}>
            <Chart
                options={options}
                series={series}
                type="bar"
                width="100%"
                height="100%"
            />
        </Box>
    );
};

export default SalesTrendChart;
